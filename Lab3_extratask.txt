clear; clc; close all;
A_amp = 4;
f     = 3;
sigma = 1;
U1    = 2.5;
U2    = 1.5;
t = 0:0.001:1;
s = A_amp*sin(2*pi*f*t) + 0.5*A_amp*cos(4*pi*f*t);
n = sigma*randn(size(t));
signal = s + n;
above_U1 = signal(abs(signal) > U1);
filtered = signal;
filtered(abs(filtered) < U2) = 0;
n_total = length(signal);
n_above = length(above_U1);
min_val = min(filtered);
max_val = max(filtered);
original = signal;
figure('Name', 'Signal representation');
subplot(1,2,1);
plot(t, original, 'b-', 'LineWidth', 1.2);
hold on;
plot(t, filtered, 'g:', 'LineWidth', 1.5);
yline(U1, '-.k', 'LineWidth', 1.2);
yline(U2, '--k', 'LineWidth', 1.2);
hold off;
xlabel('Time, t (s)', 'FontWeight', 'bold', 'FontSize', 13);
ylabel('Voltage, U (V)', 'FontWeight', 'bold', 'FontSize', 13);
title('a) Original and filtered signal with thresholds');
grid on;
ymin = min([original, filtered, U1, U2]) - 0.5;
ymax = max([original, filtered, U1, U2]) + 0.5;
axis([t(1) t(end) ymin ymax]);
legend('Original signal', 'Filtered signal', 'U_1 threshold', 'U_2 threshold', ...
       'Location', 'best');
subplot(1,2,2);
idxAboveU1 = abs(original) > U1;
tAbove = t(idxAboveU1);
sigAbove = original(idxAboveU1);
stem(tAbove, sigAbove, 'b', 'filled');
hold on;
[maxVal, maxIdx] = max(original);
[minVal, minIdx] = min(original);
plot(t(maxIdx), maxVal, 'r^', 'MarkerSize', 9, 'MarkerFaceColor', 'r');
plot(t(minIdx), minVal, 'gv', 'MarkerSize', 9, 'MarkerFaceColor', 'g');
hold off;
xlabel('Time, t (s)', 'FontWeight', 'bold', 'FontSize', 13);
ylabel('Voltage, U (V)', 'FontWeight', 'bold', 'FontSize', 13);
title('b) Signal values above U_1 (discrete format)');
grid on;
axis([t(1) t(end) min(original)-0.5 max(original)+0.5]);
legend('Signal above U_1', 'Maximum voltage', 'Minimum voltage', ...
       'Location', 'best');
