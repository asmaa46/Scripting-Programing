%% Task 2
prev = NaN;
while true
    m = input('Enter m: ');
    if m < 0
        f = m^2 + 1;
    elseif m == 0
        f = (m+1)/2;
    else
        f = m^2 - 1;
    end
    disp(['f(', num2str(m), ') = ', num2str(f)]);
    if m == prev
        break;
    end
    prev = m;
end

%%Extra task
clc;
A = [];
B = [];
while true
    a = round(rand*7);
    b = round(rand*9);
    A(end+1) = a;
    B(end+1) = b;
    if a == b
        break;
    end
end
figure;
plot(A, '-o');
hold on;
plot(B, '-s');
hold off;
legend('round(rand*7)', 'round(rand*9)');
xlabel('Iteration');
ylabel('Value');
title('Change of two random numbers');
grid on;
